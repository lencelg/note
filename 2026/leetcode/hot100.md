````cpp
vector<int> twoSum(vector<int>& nums, int target) {
    int n = nums.size(); 
    for(int i = 0; i < n; i++){
        for(int j = i + 1; j < n; j++){
            if(nums[i] + nums[j] == target){
                return {i, j};
            }
        }
    }
    return {-1, -1};
}

vector<int> twoSum(vector<int>& nums, int target) {
    unordered_map<int, int> hp;
    for(int i = 0; i < nums.size(); i++){
        if(hp.count(target - nums[i])) return {hp[target - nums[i]], i};
        else hp[nums[i]] = i;
    }
    return {-1, -1};
}
````

## 128. 最长连续序列

这里主要是要注意提前返回，不然会超时

````cpp
int longestConsecutive(vector<int>& nums) {
    unordered_set<int> hs;
    int n = nums.size();
    for_each(nums.begin(), nums.end(), [&hs](int num){hs.insert(num);});
    int ans = 0;
    for(int num: nums){
        if(hs.count(num - 1)){
            continue;
        }
        int cur = num;
        while(hs.count(num)){
            num++;
        }
        ans = max(ans, num - cur);
        // 提前返回
        if(ans > n / 2) break;
    }
    return ans;
}
````

## 283. 移动零

````cpp
void moveZeroes(vector<int>& nums) {
    int n = nums.size();
    int l = 0, r = 0;
    while(r < n){
        while(r < n && nums[r] == 0) r++;
        if(r == n) break;
        nums[l] = nums[r];
        l++;
        r++;
    }
    for(int i = l; i < n; i++){
        nums[i] = 0;
    }
}
````

````cpp
int maxArea(vector<int>& height) {
````

## 15. 三数之和

这里重点是如何判断跳过重复的集合，可以按照哈希的键的思想，两个数字决定了另外一个数字，于是可以跳过重复，当然这里还有小优化

````cpp
vector<vector<int>> threeSum(vector<int>& nums) {
    ranges::sort(nums);
    vector<vector<int>> ans;
    int n = nums.size();
    for(int i = 0; i < n - 2; i++){
        if(i - 1 >= 0 && nums[i] == nums[i - 1]) continue;
        if(nums[i] + nums[n - 1] + nums[n - 2] < 0) continue;
        if(nums[i] + nums[i + 1] + nums[i + 2] > 0) break;

        int l = i + 1, r = n - 1;
        while(l < r){
            if(nums[i] + nums[l] + nums[r] > 0) r--;
            else if(nums[i] + nums[l] + nums[r] < 0) l++;
            else{
                ans.push_back({nums[i], nums[l], nums[r]});
                l++; r--;
                while(l + 1 < n && nums[l] == nums[l - 1]) l++;
                while(r - 1 > l && nums[r] == nums[r + 1]) r--;
            }
        }
    }
    return ans;
}
````

````cpp
int trap(vector<int>& height) {
    int n = height.size(); 
    int pre_max = 0, suf_max = 0;
    int l = 0, r = n - 1;
    int ans = 0;
    while(l < r){
        pre_max = max(pre_max, height[l]);
        suf_max = max(suf_max, height[r]);
        if(pre_max < suf_max){
            ans += pre_max - height[l++];
        }else{
            ans += suf_max - height[r--];
        }
    }
    return ans;
}

int trap(vector<int>& height) {
    int ans = 0, n = height.size();
    stack<int> st;
    for(int i = 0; i < n; i++){
        while(!st.empty() && height[i] > height[st.top()]){
            int top = st.top();
            st.pop();

            if(st.empty()){
                break;
            }

            int left = st.top();
            int w = i - left - 1;
            int h = min(height[left], height[i]) - height[top];
            ans += w * h;
        }
        st.push(i);
    }
    return ans;
}
````

````cpp
int lengthOfLongestSubstring(string s) {

    int ans = 0;
    int l = 0, r = 0, n = s.size();
````

## 438. 找到字符串中所有字母的异位词

小优化：当前一次是匹配的时候并且加入的字符和上一次去除的字符相同的时候就可以判断为`true`了，不需要重新判断

````cpp
vector<int> findAnagrams(string s, string p) {
    unordered_map<int, int> hp; 
    for(char c : p){
        hp[c]++;
    }

    auto check = [&hp](void){
        for(auto& [k, v]: hp){
            if(v != 0) return false;
        }
        return true;
    };

    vector<int> ans;
    int l = 0, r = 0, n = s.size(), m = p.size();
    bool prev_true = false;
    for(; r < s.size(); r++){
        if(hp.count(s[r])){
            hp[s[r]]--;
        }

        if(r - l + 1 < m){
            continue;
        }

        if(prev_true && l && s[r] == s[l - 1]){
            ans.push_back(l);
            hp[s[l]]++;
            l++;
            continue;
        }
        if(check()){
            prev_true = true;
            ans.push_back(l);
            hp[s[l]]++;
            l++;
            continue;
        }
        prev_true = false;
        if(hp.count(s[l])) hp[s[l]]++;
        l++;
    }
    return ans;
}
````

## 239. 滑动窗口最大值

使用优先队列加懒删除

````cpp
vector<int> maxSlidingWindow(vector<int>& nums, int k) {
    // <val, index> pair
    typedef pair<int, int> pii;
    priority_queue<pii, vector<pii>> pq;
    int l = 0;
    int n = nums.size();
    vector<int> ans;
    for(int r = 0; r < n; r++){
        pq.push({nums[r], r});
        if(r - l + 1 < k){
            continue;
        }
        while(!pq.empty()){
            auto [val, idx] = pq.top();
            if(idx < l){
                pq.pop();
                continue;
            }
            ans.push_back(val);
            break;
        }

        // need to pop left but with lazy
        while(!pq.empty() && pq.top().second <= l){
            pq.pop();
        }
        l++;
    }
    return ans;
}
````

[灵神题解](https://leetcode.cn/problems/sliding-window-maximum/solutions/2499715/shi-pin-yi-ge-shi-pin-miao-dong-dan-diao-ezj6)，看了题解发现是可以用双端队列模拟，灵神的题解很好，这里要理解的点是每次只移除一个元素是因为窗口是每次只向右边移动一格的

下面的代码来自灵神，这里只做交流学习使用

````cpp
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        int n = nums.size();
        vector<int> ans(n - k + 1); // 窗口个数
        deque<int> q; // 双端队列

        for (int i = 0; i < n; i++) {
            // 1. 右边入
            while (!q.empty() && nums[q.back()] <= nums[i]) {
                q.pop_back(); // 维护 q 的单调性
            }
            q.push_back(i); // 注意保存的是下标，这样下面可以判断队首是否离开窗口

            // 2. 左边出
            int left = i - k + 1; // 窗口左端点
            if (q.front() < left) { // 队首离开窗口
                q.pop_front();
            }

            // 3. 在窗口左端点处记录答案
            if (left >= 0) {
                // 由于队首到队尾单调递减，所以窗口最大值就在队首
                ans[left] = nums[q.front()];
            }
        }

        return ans;
    }
};
````

## 287. 寻找重复数

快慢指针先找到环路，这里有一定的数学推倒，因为`slow`行进距离是环的整数倍，于是重新使用一个指针从前面走就可以走到人口了

````cpp
int findDuplicate(vector<int>& nums) {
    int slow = 0, fast = 0;
    do {
        slow = nums[slow];
        fast = nums[nums[fast]];
    }while(slow != fast);

    fast = 0;
    while(fast != slow){
        fast = nums[fast];
        slow = nums[slow];
    }
    return slow;
}
````

## 76. 最小覆盖子串

````cpp
string minWindow(string s, string t)
{
    int n = s.size(), m = t.size();
    if (m > n)
        return "";
    unordered_map<int, int> hp;
    for (auto c : t)
    {
        hp[c]++;
    }

    auto check = [&hp]() {
        for (auto &[_, v] : hp)
        {
            if (v > 0)
                return false;
        }
        return true;
    };
    int l = 0;
    int al = 0, len = INT_MAX;

    while (l < n && !hp.count(s[l]))
        l++;

    for (int r = l; r < s.size(); r++)
    {
        char cur = s[r];
        if (hp.count(cur))
            hp[cur]--;

        if (!check())
            continue;

        while (r > l)
        {
            if (hp.count(s[l]))
            {
                if (hp[s[l]] == 0)
                    break;
                hp[s[l]]++;
            }
            l++;
        }

        if (len > (r - l + 1))
        {
            al = l;
            len = r - l + 1;
        }

        hp[s[l]]++;
        l++;
    }
    return len == INT_MAX ? "" : s.substr(al, len);
}
````

## 56. 合并区间

````cpp
vector<vector<int>> merge(vector<vector<int>>& intervals) {
    sort(intervals.begin(), intervals.end()); 
    vector<vector<int>> ans;
    int pre_start = intervals[0][0], pre_end = intervals[0][1];
    for(int i = 1; i < intervals.size(); i++){
        if(intervals[i][0] <= pre_end){
            pre_end = max(pre_end, intervals[i][1]);
        }else{
            ans.push_back({pre_start, pre_end});
            pre_start = intervals[i][0];
            pre_end = intervals[i][1];
        }
    }
    ans.push_back({pre_start, pre_end});
    return ans;
}

vector<vector<int>> merge(vector<vector<int>>& intervals) {
    sort(intervals.begin(), intervals.end()); 
    vector<vector<int>> ans;
    int start = INT_MAX, end = INT_MIN;
    int n = intervals.size();
    for(int i = 0; i < n; i++){
        start = min(start, intervals[i][0]);
        end = max(end, intervals[i][1]);
        if(i == n - 1 || intervals[i + 1][0] > end){
            ans.push_back({start, end});
            start = INT_MAX;
        }
    }
    return ans;
}
````
